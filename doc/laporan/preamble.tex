\usepackage{seqsplit}

\newcommand{\ciphertextfile}[1]{%
  \par\smallskip
  \begin{tcolorbox}[breakable, colback=backcolour, colframe=black!55,
                    boxrule=0.4pt, arc=1mm, left=2mm, right=2mm, top=2mm, bottom=2mm]
  \begingroup
  \ttfamily\footnotesize\selectfont
  \endlinechar=-1\relax
  \openin\ctreadfile=#1\relax
  \loop\unless\ifeof\ctreadfile
    \readline\ctreadfile to \ctline
    \ifx\ctline\empty\else
      \expandafter\seqsplit\expandafter{\ctline}%
    \fi
  \repeat
  \closein\ctreadfile
  \endgroup
  \end{tcolorbox}%
  \par\smallskip
}
